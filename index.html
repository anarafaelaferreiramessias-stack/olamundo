import * as THREE from 'https://cdn.skypack.dev/three@0.132.2'; // --- VARIÁVEIS DO MOTOR --- let scene, camera, renderer, clock; let moveF = false, moveB = false, moveL = false, moveR = false, canJump = false; let velocity = new THREE.Vector3(); let direction = new THREE.Vector3(); // --- CARREGAR TEXTURAS PIXELADAS (IGUAL AO JAVA) --- const loader = new THREE.TextureLoader(); const textureAtlas = loader.load('https://threejs.org/examples/textures/minecraft/atlas.png'); textureAtlas.magFilter = THREE.NearestFilter; // Deixa os pixels nítidos // --- FUNÇÃO PARA INICIAR O JOGO (CHAMADA PELO BOTÃO JOGAR) --- window.iniciarJogo = function() { const launcher = document.getElementById('launcher'); if (launcher) launcher.style.display = 'none'; init(); }; function init() { // 1. Criar o Mundo scene = new THREE.Scene(); scene.background = new THREE.Color(0x7fb3d5); // Cor do céu scene.fog = new THREE.Fog(0x7fb3d5, 5, 40); camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000); camera.position.set(0, 2, 5); renderer = new THREE.WebGLRenderer({ antialias: false }); renderer.setSize(window.innerWidth, window.innerHeight); document.body.appendChild(renderer.domElement); scene.add(new THREE.AmbientLight(0xffffff, 0.8)); // 2. Criar os Materiais (Corta o Atlas para pegar Grama e Terra) const criarMaterial = (x, y) => { const tex = textureAtlas.clone(); tex.needsUpdate = true; tex.repeat.set(0.0625, 0.0625); tex.offset.set(x * 0.0625, 1 - (y + 1) * 0.0625); return new THREE.MeshLambertMaterial({ map: tex }); }; const blocoGrama = [ criarMaterial(3, 0), criarMaterial(3, 0), // Lados criarMaterial(0, 0), criarMaterial(2, 0), // Topo e Base criarMaterial(3, 0), criarMaterial(3, 0) ]; // 3. Gerar Chão de Blocos Quadrados const geo = new THREE.BoxGeometry(1, 1, 1); for (let x = -10; x < 10; x++) { for (let z = -10; z < 10; z++) { const block = new THREE.Mesh(geo, blocoGrama); block.position.set(x, 0, z); scene.add(block); } } // 4. Controles e Mouse configurarControles(); clock = new THREE.Clock(); animate(); } function configurarControles() { document.addEventListener('keydown', (e) => { if (e.code === 'KeyW') moveF = true; if (e.code === 'KeyS') moveB = true; if (e.code === 'KeyA') moveL = true; if (e.code === 'KeyD') moveR = true; if (e.code === 'Space' && canJump) { velocity.y += 10; canJump = false; } }); document.addEventListener('keyup', (e) => { if (e.code === 'KeyW') moveF = false; if (e.code === 'KeyS') moveB = false; if (e.code === 'KeyA') moveL = false; if (e.code === 'KeyD') moveR = false; }); renderer.domElement.addEventListener('click', () => { renderer.domElement.requestPointerLock(); }); document.addEventListener('mousemove', (e) => { if (document.pointerLockElement === renderer.domElement) { camera.rotation.y -= e.movementX * 0.002; camera.rotation.x -= e.movementY * 0.002; camera.rotation.x = Math.max(-1.5, Math.min(1.5, camera.rotation.x)); camera.rotation.order = 'YXZ'; } }); } function animate() { requestAnimationFrame(animate); const delta = clock.getDelta(); // Física e Movimento velocity.x -= velocity.x * 10.0 * delta; velocity.z -= velocity.z * 10.0 * delta; velocity.y -= 30.0 * delta; // Gravidade direction.z = Number(moveF) - Number(moveB); direction.x = Number(moveR) - Number(moveL); direction.normalize(); if (moveF || moveB) velocity.z -= direction.z * 100.0 * delta; if (moveL || moveR) velocity.x -= direction.x * 100.0 * delta; camera.translateX(-velocity.x * delta); camera.translateZ(-velocity.z * delta); camera.position.y += velocity.y * delta; // Colisão com o chão if (camera.position.y < 2) { velocity.y = 0; camera.position.y = 2; canJump = true; } renderer.render(scene, camera); }
