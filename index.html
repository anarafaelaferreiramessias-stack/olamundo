// --- VARIÁVEIS PARA O SISTEMA DE ANIMAÇÃO --- let swing = false; let swingTime = 0; let mouseSwayX = 0; let mouseSwayY = 0; // Atualize seu event listener de mousemove para capturar o sway document.addEventListener('mousemove', e => { if(document.pointerLockElement) { camera.rotation.y -= e.movementX * 0.002; camera.rotation.x -= e.movementY * 0.002; camera.rotation.x = Math.max(-1.5, Math.min(1.5, camera.rotation.x)); // Captura o movimento para a inércia do braço mouseSwayX = e.movementX * 0.05; mouseSwayY = e.movementY * 0.05; } }); // Adicione isso no mousedown para disparar a animação document.addEventListener('mousedown', () => { if(document.pointerLockElement) swing = true; }); function animate() { requestAnimationFrame(animate); const time = Date.now() * 0.005; const prevPos = camera.position.clone(); // Movimentação básica velocity.x -= velocity.x * 0.1; velocity.z -= velocity.z * 0.1; velocity.y -= 0.01; direction.z = Number(moveF) - Number(moveB); direction.x = Number(moveR) - Number(moveL); direction.normalize(); if (moveF || moveB) velocity.z -= direction.z * 0.08; if (moveL || moveR) velocity.x -= direction.x * 0.08; camera.translateX(-velocity.x); camera.translateZ(velocity.z); // Colisão for(let o of objects) { if(camera.position.distanceTo(o.position) < 0.8 && Math.abs(camera.position.y - o.position.y) < 1.2) { camera.position.x = prevPos.x; camera.position.z = prevPos.z; } } camera.position.y += velocity.y; if (camera.position.y < 1.6) { velocity.y = 0; camera.position.y = 1.6; canJump = true; } // --- SISTEMA DE ANIMAÇÃO DO BRAÇO (THE "FEEL") --- if(handGroup) { handGroup.position.copy(camera.position); handGroup.rotation.copy(camera.rotation); // 1. Balanço ao andar (Bobbing) let isMoving = (moveF || moveB || moveL || moveR); let bobX = isMoving ? Math.cos(time) * 0.05 : 0; let bobY = isMoving ? Math.sin(time * 2) * 0.03 : 0; // 2. Animação de Ataque (Swing) let swingZ = 0; if (swing) { swingTime += 0.2; swingZ = Math.sin(swingTime) * 0.5; // Faz o braço ir para frente e voltar if (swingTime >= Math.PI) { swingTime = 0; swing = false; } } // 3. Aplicar tudo com suavidade (Interpolando o Sway) handGroup.translateX(0.6 + bobX - (mouseSwayX * 0.01)); handGroup.translateY(-0.5 + bobY + (mouseSwayY * 0.01)); handGroup.translateZ(-0.8 - swingZ); // Rotação extra no swing para parecer um soco/picaretada handGroup.rotation.x -= swingZ * 1.2; // Reduz o sway gradualmente para o braço centralizar mouseSwayX *= 0.8; mouseSwayY *= 0.8; } renderer.render(scene, camera); }
